def area(p1, p2, p3):
    """Twice the signed area of triangle p1, p2, p3."""
    return abs((p2[0]-p1[0])*(p3[1]-p1[1]) - (p3[0]-p1[0])*(p2[1]-p1[1]))

def find_area3_triangles(grid_size):
    """Find all triangles of area 3 in a grid_size x grid_size grid."""
    points = [(x, y) for x in range(grid_size) for y in range(grid_size)]
    triangles = []
    for i in range(len(points)):
        for j in range(i+1, len(points)):
            for k in range(j+1, len(points)):
                if area(points[i], points[j], points[k]) == 6:  # 2*area = 6
                    triangles.append((i, j, k))
    return points, triangles
