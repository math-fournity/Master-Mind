from pysat.solvers import Glucose3
from pysat.card import CardEnc, EncType

def check_coloring(N, k):
    """Check if a k-coloring of {1,...,N} exists that avoids monochromatic solutions."""
    
    # Variables: c[n][i] for n in 1..N, i in 0..k-1
    # Variable number: (n-1)*k + i + 1
    def var(n, i):
        return (n-1)*k + i + 1
    
    # Generate solutions
    solutions = []
    for x in range(1, N+1):
        for y in range(1, N+1):
            for z in range(1, N+1):
                w = 2*y + 3*z - 6*x
                if 1 <= w <= N:
                    solutions.append((w, x, y, z))
    
    # Build clauses
    clauses = []
    
    # Exactly one color per number
    for n in range(1, N+1):
        # At least one
        clauses.append([var(n, i) for i in range(k)])
        # At most one
        for i in range(k):
            for j in range(i+1, k):
                clauses.append([-var(n, i), -var(n, j)])
    
    # No monochromatic solution
    for (w, x, y, z) in solutions:
        for i in range(k):
            clauses.append([-var(w, i), -var(x, i), -var(y, i), -var(z, i)])
    
    # Solve
    solver = Glucose3()
    for clause in clauses:
        solver.add_clause(clause)
    
    result = solver.solve()
    solver.delete()
    return result

# Check 2-coloring
for N in [50, 100, 200]:
    result = check_coloring(N, 2)
    print(f"2-coloring with N={N}: {'exists' if result else 'does not exist'}")
